Create a provider config

USAGE
  openshell provider create [OPTIONS] --name <NAME> --type <PROVIDER_TYPE>

FLAGS:
      --name <NAME>
          Provider name

      --type <PROVIDER_TYPE>
          Provider type

      --label <KEY=VALUE>
          Labels to attach to the provider (KEY=VALUE, repeatable)

      --from-existing
          Load provider credentials/config from existing local state

      --credential <KEY[=VALUE]>
          Provider credential pair (`KEY=VALUE`) or env lookup key (`KEY`)

      --from-gcloud-adc
          Configure credentials from gcloud Application Default Credentials (`~/.config/gcloud/application_default_credentials.json`). Valid for providers whose profile declares an ADC-compatible credential

      --from-oidc-token
          Store the active gateway OIDC access token as the named provider credential

      --runtime-credentials
          Create a provider whose required credentials are resolved at runtime by the gateway/sandbox

      --config <KEY=VALUE>
          Provider config key/value pair

      --global-profile
          Use a platform-scoped (global) provider profile instead of a workspace-scoped one

GATEWAY FLAGS:
  -g, --gateway <GATEWAY>
          Gateway name to operate on (resolved from stored metadata)
          
          [env: OPENSHELL_GATEWAY=]

      --gateway-endpoint <GATEWAY_ENDPOINT>
          Gateway endpoint URL (e.g. <https://gateway.example.com>). Connects directly without looking up gateway metadata
          
          [env: OPENSHELL_GATEWAY_ENDPOINT=]

      --gateway-insecure
          Skip TLS certificate verification for gateway connections
          
          [env: OPENSHELL_GATEWAY_INSECURE=]

GLOBAL FLAGS:
      --workspace <WORKSPACE>
          Workspace scope for resource operations
          
          [env: OPENSHELL_WORKSPACE=]
          [default: default]

  -v, --verbose...
          Increase verbosity (-v, -vv, -vvv)

      --color <WHEN>
          When to colorize output. `auto` colorizes only when stdout is a terminal

          Possible values:
          - auto:   Colorize a stream only when that stream is a terminal and no environment override applies
          - always: Always colorize, even when output is redirected
          - never:  Never colorize
          
          [env: OPENSHELL_COLOR=]
          [default: auto]

  -h, --help
          Print help

  -V, --version
          Print version
